package mathops

func Add(a, b int) int {
    return a + b
}

func AddWithLoop(a, b int) int {
    sum := 0
    for i := 0; i < b; i++ {
        sum += 1
    }
    return a + sum - b
}

package mathops

import "testing"

func BenchmarkAdd(b *testing.B) {
    for i := 0; i < b.N; i++ {
        _ = Add(10, 20)
    }
}

func BenchmarkAddWithLoop(b *testing.B) {
    for i := 0; i < b.N; i++ {
        _ = AddWithLoop(10, 20)
    }
}

func BenchmarkAdd_Sub(b *testing.B) {
    cases := []struct {
        name string
        a, c int
    }{
        {"small", 1, 2},
        {"medium", 100, 200},
        {"large", 1000, 2000},
    }

    for _, tc := range cases {
        b.Run(tc.name, func(b *testing.B) {
            for i := 0; i < b.N; i++ {
                _ = Add(tc.a, tc.c)
            }
        })
    }
}
